local function clampRestore(position)
    local size=overlay.AbsoluteSize
    if size.X<1 or size.Y<1 then return end
    local x=math.clamp(position.X,0,math.max(0,size.X-restore.AbsoluteSize.X))
    local y=math.clamp(position.Y,0,math.max(0,size.Y-restore.AbsoluteSize.Y))
    restore.Position=UDim2.fromScale(x/size.X,y/size.Y)
end
track(overlay:GetPropertyChangedSignal("AbsoluteSize"):Connect(function()
    restoreDrag=nil
    clampRestore(Vector2.new(restore.Position.X.Scale*overlay.AbsoluteSize.X+restore.Position.X.Offset,restore.Position.Y.Scale*overlay.AbsoluteSize.Y+restore.Position.Y.Offset))
end))
restore.InputBegan:Connect(function(input)
    if input.UserInputType==Enum.UserInputType.MouseButton1 or input.UserInputType==Enum.UserInputType.Touch then
        if restoreDrag then return end
        restoreBlockUntil=0
        restoreDrag={input=input,start=input.Position,origin=restore.AbsolutePosition-overlay.AbsolutePosition,moved=false}
    end
end)
track(Input.InputChanged:Connect(function(input)
    local state=restoreDrag
    if not state or not (input==state.input or state.input.UserInputType==Enum.UserInputType.MouseButton1 and input.UserInputType==Enum.UserInputType.MouseMovement) then return end
    local delta=input.Position-state.start
    if delta.Magnitude>=6 then state.moved=true end
    if state.moved then
        restoreBlockUntil=os.clock()+.3
        clampRestore(state.origin+Vector2.new(delta.X,delta.Y))
    end
end))
track(Input.InputEnded:Connect(function(input)
    local state=restoreDrag
    if not restore.Visible then restoreDrag=nil; return end
    if state and (input==state.input or state.input.UserInputType==Enum.UserInputType.MouseButton1 and input.UserInputType==Enum.UserInputType.MouseButton1) then
        if state.moved then restoreBlockUntil=os.clock()+.3 end
        restoreDrag=nil
    end
end))
track(Input.WindowFocusReleased:Connect(function()
    if restoreDrag then restoreBlockUntil=os.clock()+.3 end
    restoreDrag=nil
end))

local function dropdown(c,name,caption,options,default,full,multi,placeholder)
    local r=row(c,full and "selectFull" or "select",name,caption)
    r.label=text(r.frame,"Label",caption,16,C.muted); r.label.TextWrapped=true
    local b=button(r.frame,"Select"); b.BackgroundColor3=C.field; b.BackgroundTransparency=0; border(b,.6); hover(b); r.control=b
    local value=multi and "" or initial(name,default)
    if not multi and not table.find(options,value) then value=default; gui:SetAttribute(name,value) end
    local captionLabel=text(b,"Value",value,16,C.text); captionLabel.Position=UDim2.fromOffset(12,0); captionLabel.Size=UDim2.new(1,-45,1,0); captionLabel.TextTruncate=Enum.TextTruncate.AtEnd
    local arrow=icon(b,"chevron-down",15,C.muted); arrow.Position=UDim2.new(1,-28,.5,-7.5)
    local optionRows={}
    local ownerMenu
    local optionSet={}
    for _,option in ipairs(options) do optionSet[option]=true end
    local Http=multi and game:GetService("HttpService")
    local function selectionOf(value,allowMissing)
        if type(value)=="string" then local ok,decoded=pcall(Http.JSONDecode,Http,value); if not ok then return nil end; value=decoded end
        if type(value)~="table" then return nil end
        local selected,seen={},{}
        for key,item in pairs(value) do
            if type(key)~="number" or key%1~=0 or key<1 or key>#value or type(item)~="string" then return nil end
            if not allowMissing and not optionSet[item] then return nil end
        end
        for _,item in ipairs(value) do
            if optionSet[item] and not seen[item] then table.insert(selected,item); seen[item]=true end
        end
        return selected
    end
    local function paintSelection()
        if not multi then return end
        local selected=r.Get()
        captionLabel.Text=#selected>0 and table.concat(selected,", ") or (placeholder or "Any")
        for option,item in pairs(optionRows) do
            local on=table.find(selected,option)~=nil
            item.label.TextColor3=on and C.blue or C.text; item.check.Visible=on
        end
    end
    r.multi=multi==true
    if multi then
        r.Get=function() return selectionOf(gui:GetAttribute(name),true) or {} end
        r.Validate=function(nextValue) return selectionOf(nextValue,false)~=nil end
        r.Set=function(nextValue)
            local selected=selectionOf(nextValue,false); assert(selected,"Invalid dropdown selections")
            gui:SetAttribute(name,Http:JSONEncode(selected)); paintSelection()
        end
        r.Set(selectionOf(saved[name],true) or selectionOf(default,false) or {})
        track(gui:GetAttributeChangedSignal(name):Connect(paintSelection))
    else
        r.Set=function(nextValue)
            assert(table.find(options,nextValue),"Dropdown value is not in Values")
            captionLabel.Text=nextValue; gui:SetAttribute(name,nextValue)
        end
    end
    r.options=options
    r.Refresh=function(values)
        assert(type(values)=="table","Dropdown values must be a table")
        for _,value in ipairs(values) do assert(type(value)=="string","Dropdown values must be strings") end
        closeDropdown(); local selected=multi and r.Get()
        options=(multi or #values>0) and table.clone(values) or {"None"}; r.options=options; optionRows={}
        optionSet={}; for _,option in ipairs(options) do optionSet[option]=true end
        if multi then r.Set(selectionOf(selected,true))
        elseif not table.find(options,gui:GetAttribute(name)) then r.Set(options[1]) end
    end
    b.Activated:Connect(function()
        if openMenu and openMenu:GetAttribute("Owner")==name then closeDropdown(); return end
        closeDropdown()
        local dismiss=button(root,"DismissDropdown"); dismiss.Size=UDim2.fromScale(1,1); dismiss.ZIndex=50; dismiss.Activated:Connect(closeDropdown)
        local left,top,right,bottom=popupBounds()
        local width=math.min(right-left,math.max(b.AbsoluteSize.X/scaleFactor,208))
        local height=math.min((#options+(multi and 1 or 0))*46+62,bottom-top,320)
        local p=(b.AbsolutePosition-root.AbsolutePosition)/scaleFactor
        local y=p.Y+b.AbsoluteSize.Y/scaleFactor+6; if y+height>bottom then y=p.Y-height-6 end
        y=math.clamp(y,top,bottom-height)
        local menu=box(root,"DropdownMenu",C.hover); menu.ZIndex=60; menu.ClipsDescendants=true
        place(menu,math.clamp(p.X,left,right-width),y,width,height); corner(menu,12); border(menu,.15); menu:SetAttribute("Owner",name); openMenu=menu
        local searchField=box(menu,"SearchField",C.field); searchField.ZIndex=61; place(searchField,8,8,width-16,36); corner(searchField,8); border(searchField,.65)
        local searchGlyph=icon(searchField,"search",16,C.faint); searchGlyph.Position=UDim2.fromOffset(10,10)
        local popupSearch=make("TextBox",{Name="Search",Text="",PlaceholderText="Search options...",ClearTextOnFocus=false,BackgroundTransparency=1,TextColor3=C.text,PlaceholderColor3=C.faint,TextSize=15,FontFace=FONT_BODY,TextXAlignment=Enum.TextXAlignment.Left,ZIndex=62,Position=UDim2.fromOffset(34,0),Size=UDim2.new(1,-42,1,0)},searchField)
        local list=make("ScrollingFrame",{Name="Options",BackgroundTransparency=1,BorderSizePixel=0,ScrollBarThickness=3,ScrollBarImageColor3=C.faint,CanvasSize=UDim2.new(),ScrollingDirection=Enum.ScrollingDirection.Y,ZIndex=61,Position=UDim2.fromOffset(4,52),Size=UDim2.new(1,-8,1,-56)},menu)
        local noResults=text(list,"NoResults","No results found",14,C.muted); noResults.TextXAlignment=Enum.TextXAlignment.Center; place(noResults,4,0,width-16,44); noResults.Visible=false
        local choices={}
        optionRows={}
        ownerMenu=menu
        menu.Destroying:Connect(function() if ownerMenu==menu then optionRows={}; ownerMenu=nil end end)
        if multi then
            local clear=button(list,"ClearSelection"); clear.ZIndex=62; place(clear,0,0,width-8,44); hover(clear)
            local label=text(clear,"Label","Clear selection",14,C.muted); label.Position=UDim2.fromOffset(12,0); label.Size=UDim2.new(1,-24,1,0)
            clear.Activated:Connect(function() r.Set({}) end)
            table.insert(choices,{button=clear,label="",always=true})
        end
        if #options>40 then
            local pool,filtered={},{}
            local offset=multi and 1 or 0
            local visibleRows=math.ceil(math.max(44,height-56)/46)+2
            for index=1,visibleRows do
                local choice=button(list,"Option"..index); choice.ZIndex=62; hover(choice)
                local caption=text(choice,"Label","",14,C.text); caption.Position=UDim2.fromOffset(12,0); caption.Size=UDim2.new(1,-44,1,0)
                local check=icon(choice,"check",16,C.blue); check.Position=UDim2.new(1,-28,.5,-8)
                local item={button=choice,label=caption,check=check}
                pool[index]=item
                choice.Activated:Connect(function()
                    local option=item.option; if not option then return end
                    if multi then
                        local selected=r.Get(); local found=table.find(selected,option)
                        if found then table.remove(selected,found) else table.insert(selected,option) end
                        r.Set(selected)
                    else r.Set(option); closeDropdown() end
                end)
            end
            local function renderOptions()
                optionRows={}
                local first=math.max(1,math.floor(list.CanvasPosition.Y/46)+1-offset)
                local selected=multi and r.Get()
                for index,item in ipairs(pool) do
                    local position=first+index-1
                    local option=filtered[position]
                    item.option=option; item.button.Visible=option~=nil
                    if option then
                        place(item.button,0,(position-1+offset)*46,width-8,44)
                        item.label.Text=option
                        local on=multi and table.find(selected,option)~=nil or not multi and option==captionLabel.Text
                        item.label.TextColor3=on and C.blue or C.text; item.check.Visible=on
                        optionRows[option]=item
                    end
                end
            end
            local function searchOptions()
                local query=string.match(string.lower(popupSearch.Text),"^%s*(.-)%s*$")
                filtered={}
                for _,option in ipairs(options) do
                    if query=="" or string.find(string.lower(option),query,1,true) then table.insert(filtered,option) end
                end
                list.CanvasSize=UDim2.fromOffset(0,math.max(44,(#filtered+offset)*46)+(multi and #filtered==0 and 44 or 0))
                list.CanvasPosition=Vector2.zero
                noResults.Position=UDim2.fromOffset(4,offset*46); noResults.Visible=#filtered==0
                renderOptions()
            end
            list:GetPropertyChangedSignal("CanvasPosition"):Connect(renderOptions)
            popupSearch:GetPropertyChangedSignal("Text"):Connect(searchOptions)
            searchOptions()
            if not Input.TouchEnabled then popupSearch:CaptureFocus() end
            return
        end
        local selected=multi and r.Get()
        for i,opt in ipairs(options) do
            local choice=button(list,"Option"..i); choice.ZIndex=62; place(choice,0,(i-1)*46,width-8,44); hover(choice)
            local on=multi and table.find(selected,opt)~=nil or not multi and opt==captionLabel.Text
            local caption=text(choice,"Label",opt,14,on and C.blue or C.text); caption.Position=UDim2.fromOffset(12,0); caption.Size=UDim2.new(1,-44,1,0)
            local check=icon(choice,"check",16,C.blue); check.Position=UDim2.new(1,-28,.5,-8); check.Visible=on
            optionRows[opt]={label=caption,check=check}
            choice.Activated:Connect(function()
                if multi then
                    local nextValue=r.Get(); local index=table.find(nextValue,opt)
                    if index then table.remove(nextValue,index) else table.insert(nextValue,opt) end
                    r.Set(nextValue)
                else r.Set(opt); closeDropdown() end
            end)
            table.insert(choices,{button=choice,label=string.lower(opt)})
        end
        local function filterOptions()
            local query=string.match(string.lower(popupSearch.Text),"^%s*(.-)%s*$")
            local count,matched=0,0
            for _,item in ipairs(choices) do
                local matches=item.always or query=="" or string.find(item.label,query,1,true)~=nil
                item.button.Visible=matches
                if matches then
                    item.button.Position=UDim2.fromOffset(0,count*46); count+=1
                    if not item.always then matched+=1 end
                end
            end
            list.CanvasPosition=Vector2.zero; list.CanvasSize=UDim2.fromOffset(0,math.max(count*46,44)+(multi and matched==0 and 44 or 0))
            noResults.Position=UDim2.fromOffset(4,multi and matched==0 and 46 or 0)
            noResults.Visible=matched==0
        end
        popupSearch:GetPropertyChangedSignal("Text"):Connect(filterOptions)
        filterOptions()
        if not Input.TouchEnabled then popupSearch:CaptureFocus() end
    end)
    return r
end

local function slider(c,name,caption,default,maxValue,minValue)
    minValue=minValue or 0
    assert(maxValue>minValue,"Slider Max must be greater than Min")
    local r=row(c,"slider",name,caption)
    r.label=text(r.frame,"Label",caption,13,C.muted); r.label.TextWrapped=true; r.label.Position=UDim2.fromOffset(0,8); r.label.Size=UDim2.new(1,-73,0,36)

    local hit=button(r.frame,"Slider"); hit.Position=UDim2.fromOffset(0,42); hit.Size=UDim2.new(1,0,0,44)
    local rail=box(hit,"Rail",C.off); rail.Position=UDim2.new(0,8,.5,-2); rail.Size=UDim2.new(1,-16,0,4); corner(rail,3)
    local fill=box(rail,"Fill",C.blue); fill.Size=UDim2.fromOffset(0,4); corner(fill,3)
    local knob=box(rail,"Knob",C.text); knob.Size=UDim2.fromOffset(12,12); knob.AnchorPoint=Vector2.new(.5,.5); knob.Position=UDim2.fromScale(0,.5); corner(knob,6)
    local function set(value)
        assert(type(value)=="number","Slider value must be a number")
        if value~=value then value=minValue end
        value=math.floor(math.clamp(value,minValue,maxValue)); input.Text=tostring(value); gui:SetAttribute(name,value)
        local fraction=(value-minValue)/(maxValue-minValue)
        fill.Size=UDim2.new(fraction,0,0,4); knob.Position=UDim2.fromScale(fraction,.5)
    end
    r.Set=set
    r.min=minValue; r.max=maxValue
    set(initial(name,default)); input.FocusLost:Connect(function() set(tonumber(input.Text) or minValue) end)
    table.insert(sliders,{hit=hit,rail=rail,set=set,max=maxValue,min=minValue}); bindSlider(sliders[#sliders]); return r
end

local function rowHeight(r,width)
    if r.kind=="games" then return #r.entries==0 and 68 or math.ceil(#r.entries/2)*98 end
    if r.kind=="banner" then return #r.tags>0 and 250 or 208 end
    if r.kind=="select" then return 52 end
    if r.kind=="selectFull" then return 74 end
    if r.kind=="slider" then return 94 end
    if r.kind=="note" then
        local measured=r.measurement
        if measured and measured.width==width and measured.scale==scaleFactor
            and measured.title==r.label.Text and measured.content==r.detail.Text then return measured.height end
        rasterText(r.label,18); rasterText(r.detail,15)
        local bounds=Vector2.new(math.max(1,math.floor((width-54)*scaleFactor)-2),100000)
        local headingSize=math.max(1,math.floor(r.label.TextSize*scaleFactor+.5))
        local detailSize=math.max(1,math.floor(r.detail.TextSize*scaleFactor+.5))
        r.headingHeight=r.label.Text~="" and math.ceil(TextService:GetTextSize(r.label.Text,headingSize,Enum.Font.BuilderSansExtraBold,bounds).Y/scaleFactor)+2 or 0
        r.detailHeight=r.detail.Text~="" and math.ceil(TextService:GetTextSize(r.detail.Text,detailSize,Enum.Font.BuilderSansBold,bounds).Y/scaleFactor)+2 or 0
        local height=math.max(58,r.headingHeight+r.detailHeight+(r.headingHeight>0 and r.detailHeight>0 and 6 or 0)+36)
        r.measurement={width=width,scale=scaleFactor,title=r.label.Text,content=r.detail.Text,height=height}
        return height
    end
    if r.kind=="action" then return 46 end
    if r.kind=="actions" then return 46 end
    if r.kind=="input" then return r.multiline and 150 or 78 end
    return 42
end

local SectionMethods={}
local function titleOf(cfg)
    assert(type(cfg)=="table" and type(cfg.Title)=="string" and cfg.Title~="","A nonempty Title is required")
    return cfg.Title
end
local function controlName(cfg)
    local name=cfg.Flag or cfg.Name or cfg.Title:gsub("[^%w_]","")
    assert(type(name)=="string" and name~="" and not controls[name],"Control Flag must be unique: "..tostring(name))
    return name
end
local function bindControl(section,cfg,r)
    r.tab=section._rowTab
    local name=r.name
    controls[name]=r
    r.save=cfg.Save~=false
    local handle={Flag=name}
    function handle:Set(value) assert(r.Set,"This control has no value"); r.Set(value) end
    local function readValue() if r.Get then return r.Get() end; return gui:GetAttribute(name) end
    function handle:Get() return readValue() end
    handle.SetValue=handle.Set
    if r.Refresh then function handle:Refresh(values) r.Refresh(values) end end
    setmetatable(handle,{__index=function(_,key)
        if key=="Value" then return readValue() end
        if key=="Instance" then if r.Ensure then r.Ensure() end; return r.frame end
    end})
    if cfg.Callback and r.kind~="action" then
        assert(type(cfg.Callback)=="function","Callback must be a function")
        r.Invoke=function()
            r.LastCallbackValue=gui:GetAttribute(name)
            return pcall(cfg.Callback,readValue())
        end
        track(gui:GetAttributeChangedSignal(name):Connect(function()
            if applyingConfig or r.LastCallbackValue==gui:GetAttribute(name) then return end
            local ok,err=r.Invoke()
            if not ok then warn("[LimboUI Callback] "..tostring(err)) end
        end))
    end
    return handle
end
local function defaultOf(cfg,fallback)
    if cfg.Value~=nil then return cfg.Value end
    if cfg.Default~=nil then return cfg.Default end
    return fallback
end
function SectionMethods:Toggle(cfg)
    local title=titleOf(cfg); local value=defaultOf(cfg,false)
    assert(type(value)=="boolean","Toggle Value must be boolean")
    return bindControl(self,cfg,toggle(self._card,controlName(cfg),title,value))
end

-- Register values immediately; build the visual control only when its section is displayed.
local function deferControl(section,cfg,kind,factory)
    local title=titleOf(cfg)
    assert(cfg.Callback==nil or type(cfg.Callback)=="function","Callback must be a function")
    local name=controlName(cfg)
    cfg=table.clone(cfg)
    local c=section._card
    local r={kind=kind,name=name,caption=title,card=c,tab=section._rowTab,multi=cfg.Multi==true,
        min=cfg.Min or 0,max=cfg.Max or 100,maxLength=cfg.MaxLength or 256,options=cfg.Values}
    local Http=r.multi and game:GetService("HttpService")
    local function normalize(value,prune)
        if kind=="toggle" then assert(type(value)=="boolean","Toggle Value must be boolean"); return value end
        if kind=="slider" then
            assert(type(value)=="number" and value==value and math.abs(value)<math.huge,"Invalid slider value")
            return math.floor(math.clamp(value,r.min,r.max))
        end
        if kind=="input" then assert(type(value)=="string" and #value<=r.maxLength,"Invalid input value"); return value end
        if r.multi then
            if type(value)=="string" then value=Http:JSONDecode(value) end
            assert(type(value)=="table","Dropdown selections must be a table")
            local selected,seen={},{}
            for key,option in pairs(value) do
                assert(type(key)=="number" and key>=1 and key%1==0 and key<=#value and type(option)=="string","Invalid dropdown selections")
                assert(prune or table.find(r.options,option),"Dropdown value is not in Values")
            end
            for _,option in ipairs(value) do
                if table.find(r.options,option) and not seen[option] then table.insert(selected,option); seen[option]=true end
            end
            return Http:JSONEncode(selected)
        end
        assert(table.find(r.options,value),"Dropdown value is not in Values"); return value
    end
    if kind=="select" then
        assert(type(r.options)=="table" and (r.multi or #r.options>0),"Dropdown requires Values")
        for _,option in ipairs(r.options) do assert(type(option)=="string","Dropdown Values must be strings") end
        if cfg.FullWidth then r.kind="selectFull" end
    elseif kind=="slider" then assert(type(r.min)=="number" and type(r.max)=="number" and r.max>r.min,"Invalid slider bounds") end
    if kind~="action" then
        local fallback=kind=="toggle" and false or kind=="slider" and r.min or kind=="input" and "" or r.multi and {} or r.options and r.options[1]
        -- Boolean false must remain a valid default.
        if kind=="toggle" then fallback=false end
        local value=normalize(defaultOf(cfg,fallback),false)
        if saved[name]~=nil then local ok,restored=pcall(normalize,saved[name],true); if ok then value=restored end end
        gui:SetAttribute(name,value)
        r.Set=function(nextValue) gui:SetAttribute(name,normalize(nextValue,false)) end
        if r.multi then
            r.Get=function() return Http:JSONDecode(gui:GetAttribute(name)) end
            r.Validate=function(value) return pcall(normalize,value,false) end
        end
    end
    if kind=="select" then
        r.Refresh=function(values)
            assert(type(values)=="table","Dropdown values must be a table")
            for _,option in ipairs(values) do assert(type(option)=="string","Dropdown Values must be strings") end
            r.options=(r.multi or #values>0) and table.clone(values) or {"None"}
            cfg.Values=r.options
            if r.multi then gui:SetAttribute(name,normalize(gui:GetAttribute(name),true))
            elseif not table.find(r.options,gui:GetAttribute(name)) then r.Set(r.options[1]) end
        end
    end
    table.insert(c.rows,r)
    r.Ensure=function()
        local nativeConfig=table.clone(cfg)
        if kind~="action" then
            nativeConfig.Callback=nil
            nativeConfig.Value=r.Get and r.Get() or gui:GetAttribute(name)
            saved[name]=gui:GetAttribute(name)
        end
        controls[name]=nil
        local ok,native=pcall(factory,section,nativeConfig)
        if not ok then controls[name]=r; error(native,0) end
        local actual=controls[name]
        local index=table.find(c.rows,actual)
        if index then table.remove(c.rows,index) end
        for key,value in pairs(actual) do r[key]=value end
        if actual.Refresh then
            r.Refresh=function(values) actual.Refresh(values); r.options=actual.options end
        end
        controls[name]=r
        r.Ensure=nil
    end
    return bindControl(section,cfg,r)
end
for name,kind in pairs({Toggle="toggle",Dropdown="select",Slider="slider",Input="input",Button="action"}) do
    local factory=SectionMethods[name]
    SectionMethods[name]=function(section,cfg) return deferControl(section,cfg,kind,factory) end
end

local configPanel
function Window:ConfigPanel(tab,options)
    assert(not configPanel,"ConfigPanel already exists")
    options=options or {}
    local folder=options.Folder or ("LimboHUB/Configs/"..string.format("%.0f",game.PlaceId))
    assert(type(folder)=="string" and folder:match("^[%w_/-]+$") and not folder:find("//",1,true) and folder:sub(1,1)~="/","Invalid config folder")
    local Http=game:GetService("HttpService")
    local current,autoload="None","None"
    local section=tab:Section({Title="Config Manager",Name="ConfigManagerCard",Icon=false,Column=options.Split and 1 or 2,Mode=options.Mode})
    local profiles,externalSection
    if options.Split then
        profiles=section
        externalSection=tab:Section({Title="Import / Export",Name="ConfigTransferCard",Icon=false,Column=2,Mode=options.Mode})
    else
        local tabs=section:Tabs({Name="ConfigTabs",Values={"Profiles","Import / Export"},Value="Profiles"})
        profiles,externalSection=tabs:Tab("Profiles"),tabs:Tab("Import / Export")
    end
    local status=profiles:Paragraph({Title="Profiles",Flag="ConfigStatus",Content="Current: None | Autoload: None",Save=false})
    local name=profiles:Input({Title="Config Name",Flag="ConfigName",Placeholder="My config",MaxLength=48,Save=false})
    local selection=profiles:Dropdown({Title="Saved Configs",Flag="SelectedConfig",Values={"None"},Value="None",Save=false,Callback=function(value) if value~="None" then name:Set(value) end end})
    local external=externalSection:Input({Title="Config JSON",Flag="ConfigJSON",MultiLine=true,MaxLength=65536,Placeholder="Paste exported config JSON here...",Save=false})
    local function notice(message) feedback(message) end
    local function updateStatus()
        status.Instance.Panel.Detail.Text="Current: "..current.." | Autoload: "..autoload
        layout()
    end
    local function validName(value)
        assert(type(value)=="string" and #value>=1 and #value<=48 and value:match("^[%w_-]+$") and value~="None","Use 1-48 letters, numbers, underscores or hyphens for a config name.")
        return value
    end
    local function file(nameValue) return folder.."/"..validName(nameValue)..".json" end
    local function ensureFolder()
        assert(type(makefolder)=="function","File storage is unavailable in this executor.")
        local path=""
        for part in folder:gmatch("[^/]+") do path=path=="" and part or path.."/"..part; pcall(makefolder,path) end
    end
    local function refresh()
        local names={}
        if type(listfiles)=="function" then
            local ok,files=pcall(listfiles,folder)
            if ok and type(files)=="table" then
                for _,path in ipairs(files) do
                    local value=type(path)=="string" and path:gsub("\\","/"):match("/([%w_-]+)%.json$")
                    if value and value~="None" and #value<=48 then table.insert(names,value) end
                end
            end
        end
        table.sort(names); selection:Refresh(names)
        return names
    end
    local function collect()
        local values={}
        for flag,r in pairs(controls) do if r.save and (r.kind=="toggle" or r.kind=="select" or r.kind=="selectFull" or r.kind=="slider" or r.kind=="input") then values[flag]=gui:GetAttribute(flag) end end
        return {Version=1,Values=values}
    end
    local function apply(data)
        assert(type(data)=="table" and data.Version==1 and type(data.Values)=="table","Invalid config format.")
        local changes={}
        for flag,value in pairs(data.Values) do
            local r=controls[flag]
            if r and r.save and r.Set then
                local valid=r.kind=="toggle" and type(value)=="boolean"
                    or (r.kind=="select" or r.kind=="selectFull") and (r.multi and r.Validate(value) or not r.multi and table.find(r.options,value)~=nil)
                    or r.kind=="slider" and type(value)=="number" and value==value and math.abs(value)<math.huge and value>=r.min and value<=r.max
                    or r.kind=="input" and type(value)=="string" and #value<=r.maxLength
                assert(valid,"Invalid value for "..tostring(flag))
                table.insert(changes,{r,value})
            end
        end
        assert(#changes>0,"No compatible controls were found in this config.")
        table.sort(changes,function(a,b) return a[1].name<b[1].name end)
        local function assign(r,value)
            r.Set(value)
            if r.Invoke then
                local ok,err=r.Invoke()
                assert(ok,"Could not apply "..r.name..": "..tostring(err))
            end
        end
        assert(not applyingConfig,"A config is already loading.")
        applyingConfig=true
        local ok,err=pcall(function()
            -- Stop workers before changing filters, including toggles that already match the profile.
            for _,change in ipairs(changes) do
                local r=change[1]
                if r.kind=="toggle" then assign(r,false) end
            end
            -- Install every input before callbacks read other controls.
            for _,change in ipairs(changes) do
                if change[1].kind~="toggle" then change[1].Set(change[2]) end
            end
            for _,change in ipairs(changes) do
                local r=change[1]
                if r.kind~="toggle" and r.Invoke then
                    local applied,message=r.Invoke()
                    assert(applied,"Could not apply "..r.name..": "..tostring(message))
                end
            end
            for _,change in ipairs(changes) do
                if change[1].kind=="toggle" and change[2] then assign(change[1],true) end
            end
        end)
        applyingConfig=false
        assert(ok,err)
        for _,change in ipairs(changes) do
            if change[1].kind=="toggle" then
                assert(gui:GetAttribute(change[1].name)==change[2],"Feature could not restore its saved state: "..change[1].name)
            end
        end
        return #changes
    end
    local function decode(raw)
        assert(type(raw)=="string" and #raw>0 and #raw<=65536,"Config JSON is empty or too large.")
        return Http:JSONDecode(raw)
    end
    local function guarded(fn)
        return function(...)
            local ok,result=pcall(fn,...)
            if not ok then notice(tostring(result):gsub("^.-:%d+: ","")) end
            return ok,result
        end
    end
    local function target() local value=name:Get(); if value=="" then value=selection:Get() end; return validName(value) end
    local save=guarded(function(value)
        value=validName(value or target()); ensureFolder(); assert(type(writefile)=="function","File storage is unavailable.")
        writefile(file(value),Http:JSONEncode(collect())); current=value; refresh(); selection:Set(value); updateStatus(); notice("Config saved: "..value)
    end)
    local load=guarded(function(value)
        value=validName(value or target()); assert(type(readfile)=="function","File storage is unavailable.")
        apply(decode(readfile(file(value)))); current=value; updateStatus(); notice("Config loaded: "..value)
    end)
    local delete=guarded(function(value)
        value=validName(value or target()); assert(type(delfile)=="function","File deletion is unavailable.")
        delfile(file(value)); if current==value then current="None" end
        if autoload==value then assert(type(writefile)=="function","File storage is unavailable."); writefile(folder.."/Autoload.txt",""); autoload="None" end
        name:Set(""); refresh(); updateStatus(); notice("Config deleted: "..value)
    end)
    local setAutoload=guarded(function()
        local value=target(); assert(type(readfile)=="function" and type(writefile)=="function","File storage is unavailable.")
        decode(readfile(file(value))); ensureFolder(); writefile(folder.."/Autoload.txt",value); autoload=value; updateStatus(); notice("Autoload set: "..value)
    end)
    local clearAutoload=guarded(function()
        ensureFolder(); assert(type(writefile)=="function","File storage is unavailable."); writefile(folder.."/Autoload.txt",""); autoload="None"; updateStatus(); notice("Autoload cleared.")
    end)
    profiles:ButtonRow({Name="ConfigSaveLoad",Buttons={{Title="Save Config",Primary=true,Callback=function() save() end},{Title="Load Config",Callback=function() load() end}}})
    profiles:ButtonRow({Name="ConfigDeleteRefresh",Buttons={{Title="Delete Config",Callback=function() delete() end},{Title="Refresh List",Callback=function() refresh(); notice("Config list refreshed.") end}}})
    profiles:ButtonRow({Name="ConfigAutoload",Buttons={{Title="Set Autoload",Callback=setAutoload},{Title="Clear Autoload",Callback=clearAutoload}}})
    externalSection:ButtonRow({Name="ConfigTransfer",Buttons={
        {Title="Export Config",Callback=guarded(function()
            local raw=Http:JSONEncode(collect()); local copy=setclipboard or toclipboard
            if type(copy)=="function" and pcall(copy,raw) then notice("Config copied to clipboard.") else external:Set(raw); notice("Config JSON placed in the input.") end
        end)},
        {Title="Import Config",Callback=guarded(function() apply(decode(external:Get())); notice("Config imported.") end)},
    }})
    configPanel={Name=name,Selection=selection,External=external,Save=save,Load=load,Delete=delete,Refresh=refresh,SetAutoload=setAutoload,ClearAutoload=clearAutoload}
    function configPanel:Autoload()
        if self.tried then return end; self.tried=true
        if type(readfile)~="function" then return end
        local ok,value=pcall(readfile,folder.."/Autoload.txt")
        if ok and type(value)=="string" and value~="" then
            local valid=pcall(validName,value)
            if valid then autoload=value; updateStatus(); load(value) end
        end
    end
    function Window:SaveConfig(value) return save(value) end
    function Window:LoadConfig(value) return load(value) end
    function Window:DeleteConfig(value) return delete(value) end
    function Window:GetConfigs() return refresh() end
    refresh()
    return configPanel
end
function Window:Announce(cfg)
    assert(type(cfg)=="table" and type(cfg.Name)=="string" and #cfg.Name>0 and #cfg.Name<=128
        and type(cfg.Message)=="string" and #cfg.Message>0 and #cfg.Message<=1200,"Invalid announcement")
    if not gui.Parent then return false end
    if not self._announcementFeed then
        self._announcementFeed=make("Frame",{Name="Announcements",BackgroundTransparency=1,AnchorPoint=Vector2.new(.5,0),Position=UDim2.new(.5,0,0,52),Size=UDim2.new(1,-24,0,0),AutomaticSize=Enum.AutomaticSize.Y,ZIndex=250},overlay)
        make("UIListLayout",{Padding=UDim.new(0,6),SortOrder=Enum.SortOrder.LayoutOrder,HorizontalAlignment=Enum.HorizontalAlignment.Center},self._announcementFeed)
        self._announcementItems={}
        track(overlay:GetPropertyChangedSignal("AbsoluteSize"):Connect(function()
            for _,item in ipairs(self._announcementItems) do item.fit() end
        end))
    end
    local items=self._announcementItems
    for _,item in ipairs(items) do if cfg.Id and item.id==cfg.Id then return true end end
    while #items>=3 do table.remove(items,1).frame:Destroy() end
    local frame=make("Frame",{Name="Announcement",BackgroundTransparency=1,Size=UDim2.new(1,0,0,0),AutomaticSize=Enum.AutomaticSize.Y,ZIndex=250},self._announcementFeed)
    local row=make("Frame",{Name="Line",BackgroundTransparency=1,AnchorPoint=Vector2.new(.5,0),Position=UDim2.fromScale(.5,0),Size=UDim2.fromOffset(0,0),AutomaticSize=Enum.AutomaticSize.Y,ZIndex=250},frame)
    make("UIListLayout",{FillDirection=Enum.FillDirection.Horizontal,VerticalAlignment=Enum.VerticalAlignment.Center,Padding=UDim.new(0,7),SortOrder=Enum.SortOrder.LayoutOrder},row)
    local sender=make("TextLabel",{Name="Sender",Text=cfg.Name,RichText=false,BackgroundTransparency=1,TextColor3=Color3.fromRGB(52,165,255),TextStrokeColor3=Color3.new(0,0,0),TextStrokeTransparency=0,FontFace=FONT_STRONG,TextTruncate=Enum.TextTruncate.AtEnd,TextXAlignment=Enum.TextXAlignment.Right,LayoutOrder=1,ZIndex=251},row)
    local badge=make("Frame",{Name="VerifiedBadge",BackgroundColor3=Color3.fromRGB(40,150,255),BorderSizePixel=0,LayoutOrder=2,ZIndex=251},row)
    corner(badge,3)
    make("TextLabel",{Name="Check",Text="✓",BackgroundTransparency=1,Size=UDim2.fromScale(1,1),TextColor3=Color3.new(1,1,1),FontFace=FONT_STRONG,TextSize=18,ZIndex=252},badge)
    local message=make("TextLabel",{Name="Message",Text=cfg.Message,RichText=false,BackgroundTransparency=1,TextColor3=Color3.new(1,1,1),TextStrokeColor3=Color3.new(0,0,0),TextStrokeTransparency=0,FontFace=FONT_STRONG,TextWrapped=true,AutomaticSize=Enum.AutomaticSize.Y,TextXAlignment=Enum.TextXAlignment.Left,LayoutOrder=3,ZIndex=251},row)
    local item={id=cfg.Id,frame=frame}
    item.fit=function()
        local available=math.max(160,overlay.AbsoluteSize.X-32)
        local size=available<500 and 18 or 25
        local nameWidth=math.min(TextService:GetTextSize(cfg.Name,size,Enum.Font.BuilderSansBold,Vector2.new(10000,1000)).X+2,available*.35)
        local badgeSize=size*.8
        local textWidth=math.max(70,math.min(TextService:GetTextSize(cfg.Message,size,Enum.Font.BuilderSansBold,Vector2.new(10000,1000)).X+3,available-nameWidth-badgeSize-14))
        sender.TextSize=size; sender.Size=UDim2.fromOffset(nameWidth,size+8)
        badge.Size=UDim2.fromOffset(badgeSize,badgeSize); badge.Check.TextSize=math.floor(size*.8)
        message.TextSize=size; message.Size=UDim2.fromOffset(textWidth,0)
        row.Size=UDim2.fromOffset(nameWidth+badgeSize+textWidth+14,0)
    end
    table.insert(items,item); item.fit()
    task.delay(8,function()
        local index=table.find(items,item)
        if index then table.remove(items,index) end
        if frame.Parent then frame:Destroy() end
    end)
    return true
end

function Window:Notify(cfg)
    assert(type(cfg)=="table" and type(cfg.Content)=="string","Notify requires Content")
    feedback(cfg.Content,type(cfg.Duration)=="number" and math.clamp(cfg.Duration,.1,60) or 3)
end
function Window:Show()
    assert(gui.Parent,"Window has been destroyed")
    assert(#navItems>0,"Create at least one tab before Show")
    if not pageNames[activePage] then activePage=navItems[1][1] end
    rebuildFeatureIndex(); ready=true; show(true); layout()
    if configPanel then configPanel:Autoload() end
    return self
end
function Window:Hide() show(false) end
function Window:Destroy() gui:Destroy() end
Window.CreateTab=Window.Tab
Window.AddTab=Window.Tab
currentWindow=Window
gui.Destroying:Connect(function()
    for _,c in ipairs(connections) do c:Disconnect() end
    if fpsWorker then pcall(task.cancel,fpsWorker); fpsWorker=nil end
end)
root.Visible=false; canvas.Visible=false

return Window
end
function Limbo:Notify(config)
    assert(currentWindow and currentWindow.Gui.Parent,"Create a window before Notify")
    currentWindow:Notify(config)
end
function Limbo:Destroy()
    if currentWindow then currentWindow:Destroy(); currentWindow=nil end
end
return Limbo
